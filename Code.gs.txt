// ========== USER AUTHENTICATION ==========
function authenticateUser(username, password) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Users');
  if (!sheet) {
    return { success: false, error: 'User database not found' };
  }
  
  const data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) {
    return { success: false, error: 'No users found' };
  }
  
  const headers = data[0];
  const userIdCol = headers.indexOf('User ID');
  const usernameCol = headers.indexOf('Username');
  const passwordCol = headers.indexOf('Password');
  const statusCol = headers.indexOf('Status');
  
  if (usernameCol === -1 || passwordCol === -1 || statusCol === -1) {
    return { success: false, error: 'Invalid Users sheet format' };
  }
  
  const normalizedUsername = username.trim().toLowerCase();
  
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const sheetUsername = (row[usernameCol] || '').trim().toLowerCase();
    const sheetPassword = (row[passwordCol] || '').trim();
    const sheetStatus = (row[statusCol] || '').trim();
    
    if (sheetUsername === normalizedUsername && sheetPassword === password) {
      if (sheetStatus !== 'Active') {
        return { success: false, error: 'Account is not active' };
      }
      return {
        success: true,
        user: {
          userId: row[userIdCol] || '',
          username: sheetUsername,
          status: sheetStatus
        }
      };
    }
  }
  
  return { success: false, error: 'Invalid username or password' };
}

// ========== HELPER FUNCTIONS ==========
const isValid = (val) => { const n = Number(val); return !isNaN(n) && n > 0.0001; };

function getCategoryCriteria() {
  if (CATEGORY_CRITERIA) return CATEGORY_CRITERIA;

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dropSheet = ss.getSheetByName('Drop');
  if (!dropSheet) return getDefaultCategories();

  let range;
  try { range = ss.getRangeByName('Category'); } catch (e) { range = null; }

  if (!range) {
    const data = dropSheet.getDataRange().getValues();
    const headerRow = data.findIndex(row => row.includes('DatefROM') && row.includes('DateTo'));
    if (headerRow === -1) return getDefaultCategories();
    const headers = data[headerRow];
    const minCol = headers.indexOf('DatefROM');
    const maxCol = headers.indexOf('DateTo');
    const actualCol = headers.indexOf('Category_Actual');
    const probableCol = headers.indexOf('Category_Probable');
    if (minCol === -1 || maxCol === -1 || actualCol === -1 || probableCol === -1) return getDefaultCategories();
    const rows = data.slice(headerRow + 1).filter(row => row[minCol] !== "" && row[maxCol] !== "");
    const criteria = rows.map(row => ({
      min: parseInt(row[minCol]) || 0,
      max: parseInt(row[maxCol]) || 9999999,
      actual: String(row[actualCol] || '').trim() || 'Pass',
      probable: String(row[probableCol] || '').trim() || 'Pass'
    })).sort((a, b) => a.min - b.min);
    CATEGORY_CRITERIA = criteria;
    return criteria;
  }

  const values = range.getValues();
  const headers = values[0];
  const minCol = headers.indexOf('DatefROM');
  const maxCol = headers.indexOf('DateTo');
  const actualCol = headers.indexOf('Category_Actual');
  const probableCol = headers.indexOf('Category_Probable');
  if (minCol === -1 || maxCol === -1 || actualCol === -1 || probableCol === -1) return getDefaultCategories();
  const criteria = values.slice(1).filter(row => row[minCol] !== "" && row[maxCol] !== "").map(row => ({
    min: parseInt(row[minCol]) || 0,
    max: parseInt(row[maxCol]) || 9999999,
    actual: String(row[actualCol] || '').trim() || 'Pass',
    probable: String(row[probableCol] || '').trim() || 'Pass'
  })).sort((a, b) => a.min - b.min);
  CATEGORY_CRITERIA = criteria;
  return criteria;
}

function getDefaultCategories() {
  return CATEGORY_CONFIG.rules;
}

function getSheetData() {
  if (GLOBAL_DATA) return GLOBAL_DATA;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Data");
  if (!sheet) throw new Error("Sheet 'Data' not found.");
  const range = sheet.getDataRange();
  const rawValues = range.getValues();
  const headers = rawValues[0].map(h => String(h).trim().toUpperCase().replace(/\s+/g, '_'));
  const data = [];
  for (let i = 1; i < rawValues.length; i++) {
    let obj = {};
    headers.forEach((header, j) => {
      var rawVal = rawValues[i][j];
      if (["OVD_DATE", "NEPALI_OVD_MONTH", "NEPALI_OVD_YEAR", "CIF_ID", "ACCOUNT_NO", "CUSTOMER_GROUP", "RENEWAL", "CATEGORY_ACTUAL", "CATEGORY_PROBABLE"].includes(header)) {
        if (rawVal instanceof Date) {
          obj[header] = rawVal;
        } else {
          var val = (rawVal === null || rawVal === undefined) ? "" : String(rawVal).trim();
          if (header === "CIF_ID") val = val.replace(/^0+/, '');
          obj[header] = val;
        }
      } else {
        obj[header] = rawVal;
      }
    });
    data.push(obj);
  }
  GLOBAL_DATA = data;
  return data;
}

function getCustomerList() {
  try {
    const full = loadFullData();
    if (full && full.allMetrics) {
      return full.allMetrics.map(m => {
        return {
          id: m.CIF_ID,
          name: m.CUSTOMER_NAME,
          metrics: m
        };
      });
    }
  } catch (e) {
    console.error(e);
  }
  
  const data = getSheetData();
  const lookup = new Map();
  data.forEach(row => {
    const rawCif = getRowValueInsensitive(row, 'CIF_ID');
    const name = String(getRowValueInsensitive(row, 'CUSTOMER_NAME') || '').trim();
    const cif = String(rawCif || '').trim().replace(/^0+/, '');
    if (cif && name && !lookup.has(cif)) {
      lookup.set(cif, { id: cif, name: name });
    }
  });
  return Array.from(lookup.values());
}

function getCustomerMetrics(cifId) {
  const data = getSheetData();
  const records = data.filter(r => r.CIF_ID === cifId);
  if (records.length === 0) return null;

  const first = records[0];
  let maxOvdDays = 0, maxRevOvdDays = 0;
  let netOverdueForCategory = 0;  // Always includes P_OVD for category calculation
  let netOverdueForDisplay = 0;    // Excludes P_OVD for EXPIRED accounts
  
  records.forEach(r => {
    const ovd = Number(r.OVD_DAYS) || 0;
    const rev = Number(r.REV_OVD_DAYS) || 0;
    if (ovd > maxOvdDays) maxOvdDays = ovd;
    if (rev > maxRevOvdDays) maxRevOvdDays = rev;
    
    const iOvd = Number(r.I_OVD) || 0;
    const pOvd = Number(r.P_OVD) || 0;
    const renewal = String(r.RENEWAL || '').trim().toUpperCase();
    
    // For category calculation: P_OVD inclusion controlled by CATEGORY_CONFIG
    netOverdueForCategory += iOvd + (CATEGORY_CONFIG.includePOVDForExpired || renewal !== 'EXPIRED' ? pOvd : 0);
    
    // For display calculation: Exclude P_OVD if EXPIRED
    if (renewal === 'EXPIRED') {
      netOverdueForDisplay += iOvd;  // Only add I_OVD, exclude P_OVD
    } else {
      netOverdueForDisplay += iOvd + pOvd;  // Add both
    }
  });
  
  const criteria = getCategoryCriteria();
  let actualCat = "Pass", probableCat = "Pass";
  
  // Use netOverdueForCategory (includes P_OVD always) for category determination
  if (Math.abs(netOverdueForCategory) < 0.0001) {
    actualCat = "Pass";
    probableCat = "Pass";
  } else {
    for (let rule of criteria) {
      if (maxOvdDays >= rule.min && maxOvdDays <= rule.max) actualCat = rule.actual;
      if (maxRevOvdDays >= rule.min && maxRevOvdDays <= rule.max) probableCat = rule.probable;
    }
  }
  
  let metrics = {
    CUSTOMER_NAME: first.CUSTOMER_NAME,
    CIF_ID: first.CIF_ID,
    CUSTOMER_GROUP: (first.CUSTOMER_GROUP || "NO GROUP").trim(),
    CATEGORY_ACTUAL: actualCat,
    CATEGORY_PROBABLE: probableCat,
    RENEWAL: first.RENEWAL || "N/A",
    TOT_OVD_AMT: 0,
    MIN_BAL_NPA: 0,
    MIN_BAL_WL: 0,
    MAX_OVD_DAYS: maxOvdDays,
    OP_BALANCE: 0,
    OUTSTANDING_LOAN: 0,
    TOTAL_OVERDUE_NET: 0,
    NET_OVERDUE: netOverdueForDisplay,  // Use display version for UI
    accounts: []
  };

  const accMap = new Map();
  records.forEach(r => {
    const accNo = r.ACCOUNT_NO;
    if (!accNo) return;
    if (!accMap.has(accNo)) {
      accMap.set(accNo, {
        ACCOUNT_NO: accNo,
        SCHEME: r.SCHEME || "N/A",
        OUTSTANDING_LOAN: 0,
        FULL_RATE: r.FULL_RATE || 0,
        MIN_BAL_NPA: 0,
        MIN_BAL_WL: 0,
        SUM_TOT_OVD: 0,
        SUM_PENAL: 0,
        NET_OVD: 0,
        RENEWAL: String(r.RENEWAL || '').trim().toUpperCase()
      });
    }
    let a = accMap.get(accNo);
    const rowLoan = Number(r.OUTSTANDING_LOAN) || 0;
    if (a.OUTSTANDING_LOAN <= 0.0001 && rowLoan > 0.0001) a.OUTSTANDING_LOAN = rowLoan;
    if (isValid(r.MIN_BAL_NPA)) a.MIN_BAL_NPA = Math.max(a.MIN_BAL_NPA, Number(r.MIN_BAL_NPA));
    if (isValid(r.MIN_BAL_WL)) a.MIN_BAL_WL = Math.max(a.MIN_BAL_WL, Number(r.MIN_BAL_WL));
    
    // For TOT_OVD_AMT: Exclude P_OVD if EXPIRED
    const rowOvdAmt = Number(r.TOT_OVD_AMT) || 0;
    const iOvd = Number(r.I_OVD) || 0;
    const pOvd = Number(r.P_OVD) || 0;
    const renewal = a.RENEWAL;
    
    let adjustedOvdAmt = rowOvdAmt;
    if (renewal === 'EXPIRED') {
      adjustedOvdAmt = rowOvdAmt - pOvd;  // Subtract P_OVD from total
    }
    
    if (isValid(adjustedOvdAmt)) a.SUM_TOT_OVD += adjustedOvdAmt;
    
    const rowPenal = Number(r.PENAL_INT) || 0;
    if (isValid(rowPenal)) a.SUM_PENAL += rowPenal;
  });

  const opRow = records.find(r => isValid(r.OP_BALANCE));
  metrics.OP_BALANCE = opRow ? Number(opRow.OP_BALANCE) : 0;

  metrics.accounts = Array.from(accMap.values()).map(a => {
    a.NET_OVD = a.SUM_TOT_OVD - a.SUM_PENAL;
    metrics.OUTSTANDING_LOAN += a.OUTSTANDING_LOAN;
    metrics.TOTAL_OVERDUE_NET += a.NET_OVD;
    metrics.TOT_OVD_AMT += a.SUM_TOT_OVD;
    metrics.MIN_BAL_NPA += a.MIN_BAL_NPA;
    metrics.MIN_BAL_WL += a.MIN_BAL_WL;
    return a;
  });

  return metrics;
}

function getGroupDetails(groupName) {
  const data = getSheetData();
  const trimmedGroup = String(groupName).trim();
  const groupRecords = data.filter(r => String(r.CUSTOMER_GROUP).trim() === trimmedGroup);
  if (groupRecords.length === 0) return null;

  let summary = { groupName: trimmedGroup, totalLoan: 0, totalOvd: 0, totalNetOvd: 0, minNpa: 0, minWl: 0, unitCount: 0 };
  const cifMap = new Map();
  const criteria = getCategoryCriteria();

  groupRecords.forEach(r => {
    if (!cifMap.has(r.CIF_ID)) {
      cifMap.set(r.CIF_ID, {
        name: r.CUSTOMER_NAME, 
        actual: '',
        probable: '',
        groupName: r.CUSTOMER_GROUP, 
        records: []
      });
    }
    cifMap.get(r.CIF_ID).records.push(r);
  });

  summary.unitCount = cifMap.size;
  const units = [];

  cifMap.forEach((cifData, cifId) => {
    let u_loan = 0, u_ovd = 0, u_penal = 0, u_npa = 0, u_wl = 0;
    let accMap = new Map();
    
    let maxOvdDays = 0;
    let maxRevOvdDays = 0;
    let netOverdueForCategory = 0;  // Always includes P_OVD
    let netOverdueForDisplay = 0;    // Excludes P_OVD for EXPIRED

    cifData.records.forEach(r => {
      const accNo = r.ACCOUNT_NO || "UNKNOWN";
      const renewal = String(r.RENEWAL || '').trim().toUpperCase();
      
      if (!accMap.has(accNo)) {
        accMap.set(accNo, { 
          loan: 0, ovd: 0, penal: 0, npa: 0, wl: 0, 
          renewal: renewal,
          iOvdTotal: 0,
          pOvdTotal: 0
        });
      }
      let a = accMap.get(accNo);
     
      const rLoan = Number(r.OUTSTANDING_LOAN) || 0;
      if (a.loan <= 0.0001 && rLoan > 0.0001) a.loan = rLoan;
      
      // Track I_OVD and P_OVD separately
      const iOvd = Number(r.I_OVD) || 0;
      const pOvd = Number(r.P_OVD) || 0;
      a.iOvdTotal += iOvd;
      a.pOvdTotal += pOvd;
      
      // For display OVD: Exclude P_OVD if EXPIRED
      const totOvdAmt = Number(r.TOT_OVD_AMT) || 0;
      if (renewal === 'EXPIRED') {
        if (isValid(totOvdAmt - pOvd)) a.ovd += (totOvdAmt - pOvd);
      } else {
        if (isValid(totOvdAmt)) a.ovd += totOvdAmt;
      }
      
      if (isValid(r.PENAL_INT)) a.penal += Number(r.PENAL_INT);
      if (isValid(r.MIN_BAL_NPA)) a.npa = Math.max(a.npa, Number(r.MIN_BAL_NPA));
      if (isValid(r.MIN_BAL_WL)) a.wl = Math.max(a.wl, Number(r.MIN_BAL_WL));
      
      const ovd = Number(r.OVD_DAYS) || 0;
      const revOvd = Number(r.REV_OVD_DAYS) || 0;
      if (ovd > maxOvdDays) maxOvdDays = ovd;
      if (revOvd > maxRevOvdDays) maxRevOvdDays = revOvd;
      
      // Category calculation: P_OVD inclusion controlled by CATEGORY_CONFIG
      netOverdueForCategory += iOvd + (CATEGORY_CONFIG.includePOVDForExpired || renewal !== 'EXPIRED' ? pOvd : 0);
      
      // Display calculation: Exclude P_OVD if EXPIRED
      if (renewal === 'EXPIRED') {
        netOverdueForDisplay += iOvd;
      } else {
        netOverdueForDisplay += iOvd + pOvd;
      }
    });

    accMap.forEach(a => { 
      u_loan += a.loan; 
      u_ovd += a.ovd; 
      u_penal += a.penal; 
      u_npa += a.npa; 
      u_wl += a.wl; 
    });
    const u_netOvd = u_ovd - u_penal;

    summary.totalLoan += u_loan; 
    summary.totalOvd += u_ovd; 
    summary.totalNetOvd += u_netOvd;
    summary.minNpa += u_npa; 
    summary.minWl += u_wl;

    let actualCat = "Pass";
    let probableCat = "Pass";
    
    // Use netOverdueForCategory for category determination
    if (Math.abs(netOverdueForCategory) < 0.0001) {
      actualCat = "Pass";
      probableCat = "Pass";
    } else {
      for (let rule of criteria) {
        if (maxOvdDays >= rule.min && maxOvdDays <= rule.max) actualCat = rule.actual;
        if (maxRevOvdDays >= rule.min && maxRevOvdDays <= rule.max) probableCat = rule.probable;
      }
    }

    units.push({
      name: cifData.name, 
      actual: actualCat, 
      probable: probableCat, 
      groupName: cifData.groupName,
      totalOvd: u_ovd, 
      totalNetOvd: u_netOvd, 
      minNpa: u_npa, 
      minWl: u_wl,
      maxOvdDays: maxOvdDays,
      netOverdue: netOverdueForDisplay  // Display version for UI
    });
  });

  return { summary, units };
}

function parseDateRobustly(val) {
  if (!val) return null;
  if (val instanceof Date) return val;
  
  // Try converting serial number (Excel/Google Sheets serial date)
  var num = Number(val);
  if (!isNaN(num) && num > 1000 && num < 100000) {
    var d = new Date((num - 25569) * 24 * 3600 * 1000);
    if (!isNaN(d.getTime()) && d.getFullYear() > 1900 && d.getFullYear() < 2200) return d;
  }
  
  // Try standard constructor
  var d = new Date(val);
  if (!isNaN(d.getTime())) return d;
  
  // Try splitting by slash, dash or dot
  var parts = String(val).split(/[\/\-\.]/);
  if (parts.length === 3) {
    var p0 = parseInt(parts[0], 10);
    var p1 = parseInt(parts[1], 10);
    var p2 = parseInt(parts[2], 10);
    
    if (p0 > 1000) {
      d = new Date(p0, p1 - 1, p2);
      if (!isNaN(d.getTime())) return d;
    } else if (p2 > 1000) {
      if (p0 > 12) {
        d = new Date(p2, p1 - 1, p0);
      } else {
        d = new Date(p2, p0 - 1, p1);
        if (isNaN(d.getTime())) {
          d = new Date(p2, p1 - 1, p0);
        }
      }
      if (!isNaN(d.getTime())) return d;
    }
  }
  return null;
}

  const records = data.filter(r => r.CIF_ID === cifId);

    const renewal = String(r.RENEWAL || '').trim().toUpperCase();
    const pOvd = Number(r.P_OVD) || 0;
    const iOvd = Number(r.I_OVD) || 0;
    const totOvdAmt = Number(r.TOT_OVD_AMT) || 0;
    const penal = Number(r.PENAL_INT) || 0;
    
    // For EXPIRED accounts, exclude P_OVD from TOT_OVD_AMT
    let adjustedTotOvd = totOvdAmt;
    let adjustedPOvd = pOvd;
    
    if (renewal === 'EXPIRED') {
      adjustedTotOvd = totOvdAmt - pOvd;
      adjustedPOvd = 0;  // Don't count P_OVD separately either
    }
    
    if (!isValid(adjustedTotOvd)) return;
    
    let dateObj = parseDateRobustly(r.OVD_DATE);
    
    if (!dateObj || isNaN(dateObj.getTime())) {
      return;
    }
    
    const year = dateObj.getFullYear();
    const monthIndex = dateObj.getMonth();
    const monthNames = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
    const monthLabel = monthNames[monthIndex] + " " + year;
    const sortKey = year + "-" + String(monthIndex + 101).substring(1);
    
    if (!group[sortKey]) {
      group[sortKey] = { OVD_DATE: monthLabel, sortKey: sortKey, TOT_OVD_AMT: 0, I_OVD: 0, P_OVD: 0, PENAL_INT: 0 };
    }
    
    group[sortKey].TOT_OVD_AMT += adjustedTotOvd;
    group[sortKey].I_OVD += iOvd;
    group[sortKey].P_OVD += adjustedPOvd;
    group[sortKey].PENAL_INT += penal;
    
    grandTotal.TOT_OVD_AMT += adjustedTotOvd;
    grandTotal.I_OVD += iOvd;
    grandTotal.P_OVD += adjustedPOvd;
    grandTotal.PENAL_INT += penal;
  });
  
  const sortedRows = Object.values(group).sort((a, b) => a.sortKey.localeCompare(b.sortKey));
  sortedRows.forEach(r => delete r.sortKey);
  
  return { rows: sortedRows, total: grandTotal };
}

function getSlicerOptions(cifId) {
  const data = getSheetData();
  const records = data.filter(r => r.CIF_ID === cifId && isValid(r.TOT_OVD_AMT));
  let yearsSet = new Set();
  let monthsSet = new Set();
  records.forEach(r => {
    if (r.NEPALI_OVD_YEAR) yearsSet.add(String(r.NEPALI_OVD_YEAR));
    if (r.NEPALI_OVD_MONTH) monthsSet.add(String(r.NEPALI_OVD_MONTH));
  });
  const sortedYears = Array.from(yearsSet).sort().reverse();
  const sortedMonths = Array.from(monthsSet).sort((a, b) => NEPALI_MONTHS.indexOf(a) - NEPALI_MONTHS.indexOf(b));
  return { years: sortedYears, months: sortedMonths };
}

function getSlicerSummary(cifId, yearsArray, months) {
  const data = getSheetData();
  const filtered = data.filter(r => {
    const rYear = String(r.NEPALI_OVD_YEAR);
    const isYear = yearsArray.includes("ALL") || yearsArray.includes(rYear);
    return r.CIF_ID === cifId && isYear && months.includes(r.NEPALI_OVD_MONTH) && isValid(r.TOT_OVD_AMT);
  });
  const group = {};
  
  filtered.forEach(r => {
    const renewal = String(r.RENEWAL || '').trim().toUpperCase();
    const pOvd = Number(r.P_OVD) || 0;
    const iOvd = Number(r.I_OVD) || 0;
    const totOvdAmt = Number(r.TOT_OVD_AMT) || 0;
    const penal = Number(r.PENAL_INT) || 0;
    
    // For EXPIRED accounts, exclude P_OVD
    let adjustedTotOvd = totOvdAmt;
    let adjustedPOvd = pOvd;
    
    if (renewal === 'EXPIRED') {
      adjustedTotOvd = totOvdAmt - pOvd;
      adjustedPOvd = 0;
    }
    
    let m = r.NEPALI_OVD_MONTH; 
    if (!group[m]) group[m] = { NEPALI_OVD_Month: m, TOT_OVD_AMT: 0, I_OVD: 0, P_OVD: 0, PENAL_INT: 0 };
    
    group[m].TOT_OVD_AMT += adjustedTotOvd;
    group[m].I_OVD += iOvd;
    group[m].P_OVD += adjustedPOvd;
    group[m].PENAL_INT += penal;
  });
  
  return Object.values(group).sort((a,b) => NEPALI_MONTHS.indexOf(a.NEPALI_OVD_Month) - NEPALI_MONTHS.indexOf(b.NEPALI_OVD_Month));
}

function getCustomerInfoData() {
  try {
    const fullData = loadFullData();
    if (!fullData || !fullData.allMetrics) return [];
    
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const customerInfoSheet = ss.getSheetByName('CustomerInfo');
    const contactsMap = new Map();
    if (customerInfoSheet) {
      const ciData = customerInfoSheet.getDataRange().getDisplayValues();
      if (ciData.length > 1) {
        const headers = ciData[0].map(h => String(h).trim());
        const cifIdx = headers.findIndex(h => {
          const norm = h.toLowerCase().replace(/[\s_-]/g, '');
          return norm === 'cifid' || norm === 'cif' || norm.indexOf('cif') !== -1;
        });
        const contactNoIdx = headers.findIndex(h => h.toLowerCase().replace(/[\s_-]/g, '') === 'contactno');
        const addContactNoIdx = headers.findIndex(h => h.toLowerCase().replace(/[\s_-]/g, '') === 'additionalcontactnumber');
        if (cifIdx !== -1) {
          for (let i = 1; i < ciData.length; i++) {
            const row = ciData[i];
            const cif = String(row[cifIdx]).trim().replace(/^0+/, '');
            contactsMap.set(cif, {
              Contact_No: contactNoIdx !== -1 ? String(row[contactNoIdx]).trim() : '',
              Additional_Contact_Number: addContactNoIdx !== -1 ? String(row[addContactNoIdx]).trim() : ''
            });
          }
        }
      }
    }
    
    return fullData.allMetrics.map(m => {
      const cifStr = String(m.CIF_ID).trim().replace(/^0+/, '');
      const contactInfo = contactsMap.get(cifStr) || { Contact_No: '', Additional_Contact_Number: '' };
      return {
        CUSTOMER_NAME: m.CUSTOMER_NAME || 'N/A',
        CIF_ID: m.CIF_ID,
        OUTSTANDING_LOAN: m.OUTSTANDING_LOAN || 0,
        CUSTOMER_GROUP: m.CUSTOMER_GROUP || 'NO GROUP',
        Contact_No: contactInfo.Contact_No,
        Additional_Contact_Number: contactInfo.Additional_Contact_Number
      };
    });
  } catch (e) {
    console.error(e);
    return [];
  }
}

function getRowValueInsensitive(row, keyToFind) {
  const normalizedKey = keyToFind.toLowerCase().replace(/[\s_-]/g, '');
  for (let key in row) {
    if (key.toLowerCase().replace(/[\s_-]/g, '') === normalizedKey) {
      return row[key];
    }
  }
  return null;
}

// ========== LOAD FULL DATA (ONE REQUEST) ==========
function loadFullData() {
  const startTime = new Date().getTime();
  
  function findColumnIndex(headers, exactMatches, containsMatches, excludes) {
    for (var ei = 0; ei < exactMatches.length; ei++) {
      var idx = headers.findIndex(function(h) {
        var norm = String(h).toLowerCase().replace(/[\s_-]/g, '');
        if (excludes) {
          for (var xi = 0; xi < excludes.length; xi++) {
            if (norm === excludes[xi]) return false;
          }
        }
        return norm === exactMatches[ei];
      });
      if (idx !== -1) return idx;
    }
    for (var ci = 0; ci < containsMatches.length; ci++) {
      var idx = headers.findIndex(function(h) {
        var norm = String(h).toLowerCase().replace(/[\s_-]/g, '');
        if (excludes) {
          for (var xi = 0; xi < excludes.length; xi++) {
            if (norm === excludes[xi]) return false;
          }
        }
        return norm.indexOf(containsMatches[ci]) !== -1;
      });
      if (idx !== -1) return idx;
    }
    return -1;
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const customerInfoSheet = ss.getSheetByName('CustomerInfo');
  const cifToRm = new Map();
  const cifToRel = new Map();
  if (customerInfoSheet) {
    const ciData = customerInfoSheet.getDataRange().getValues();
    Logger.log('CustomerInfo rows: ' + ciData.length);
    if (ciData.length > 1) {
      const ciHeaders = ciData[0].map(function(h) { return String(h).trim(); });
      Logger.log('CustomerInfo headers: ' + JSON.stringify(ciHeaders));
      var cifIdx = findColumnIndex(ciHeaders, ['cifid', 'cif', 'cid', 'customerid', 'customercode', 'cifnumber'], ['cif', 'customerid', 'accountno'], ['customername', 'customergroup', 'customerstatus']);
      var rmIdx = findColumnIndex(ciHeaders, ['rm', 'assignedrm', 'assignedto', 'representative', 'staffname', 'officername', 'employeename', 'relationshipmanager', 'rmname'], ['rm', 'relationshipmanager', 'staff', 'officer', 'assigned', 'employee', 'representative'], []);
      var relIdx = findColumnIndex(ciHeaders, ['relationship', 'custrelationship', 'customertype', 'type', 'branch'], ['relationship', 'custrelationship', 'branchlrd'], ['customername']);
      Logger.log('cifIdx=' + cifIdx + ' rmIdx=' + rmIdx + ' relIdx=' + relIdx);
      if (cifIdx !== -1) {
        var sampleCount = 0;
        for (let i = 1; i < ciData.length; i++) {
          const row = ciData[i];
          const rawCif = String(row[cifIdx]).trim();
          // Normalize: strip C/R prefix and leading zeros for consistent matching
          const normalizedCif = rawCif.replace(/^[CR]/i, '').replace(/^0+/, '');
          if (!normalizedCif) continue;
          if (rmIdx !== -1) {
            const rm = String(row[rmIdx]).trim();
            if (rm) {
              cifToRm.set(normalizedCif, rm);
              if (sampleCount < 3) { Logger.log('Sample RM: normalizedCif=' + normalizedCif + ' rm=' + rm); sampleCount++; }
            }
          }
          if (relIdx !== -1) {
            const rel = String(row[relIdx]).trim();
            if (rel) cifToRel.set(normalizedCif, rel);
          }
        }
        Logger.log('cifToRm size (normalized): ' + cifToRm.size);
        if (cifToRm.size === 0) Logger.warn('cifToRm EMPTY after processing all CustomerInfo rows');
      } else {
        Logger.warn('CIF column NOT FOUND in CustomerInfo headers');
      }
    } else {
      Logger.warn('CustomerInfo has only ' + ciData.length + ' row(s) ΓÇö header only or empty');
    }
  } else {
    Logger.warn('CustomerInfo sheet NOT FOUND in spreadsheet');
  }

  const data = getSheetData();
  const criteria = getCategoryCriteria();
  
  const customerMap = new Map();
  const groupMap = new Map();
  
  var dataCifCount = 0;
  var rmMatchCount = 0;
  data.forEach(row => {
    const cif = String(row.CIF_ID || '').trim().replace(/^0+/, '');
    if (!cif) return;
    if (dataCifCount < 5) { Logger.log('Data row CIF: ' + cif); dataCifCount++; }
    
    const renewal = String(row.RENEWAL || '').trim().toUpperCase();
    const isExpired = renewal === 'EXPIRED';
    
    if (!customerMap.has(cif)) {
      var relFromCi = cifToRel.get(cif.replace(/^[CR]/i, '').replace(/^0+/, ''));
      var relFromData = row.CUST_RELATIONSHIP || row.RELATIONSHIP || getRowValueInsensitive(row, 'relationship') || getRowValueInsensitive(row, 'custrelationship') || '';
      var rmNormalized = cif.replace(/^[CR]/i, '').replace(/^0+/, '');
      var matchedRm = cifToRm.get(rmNormalized) || 'Unassigned';
      if (matchedRm !== 'Unassigned') rmMatchCount++;
      customerMap.set(cif, {
        cifId: cif,
        name: row.CUSTOMER_NAME || 'N/A',
        group: row.CUSTOMER_GROUP || "NO GROUP",
        renewal: renewal,
        relationship: relFromData || relFromCi || 'Branch',
        assignedRM: matchedRm,
        maxOvdDays: 0,
        maxRevOvdDays: 0,
        outstandingLoan: 0,
        netOverdueForCategory: 0,  // Always includes P_OVD
        netOverdueForDisplay: 0,    // Excludes P_OVD if EXPIRED
        minBalNpa: 0,
        minBalWl: 0,
        opBalance: 0,
        accounts: new Map()
      });
    }
    
    const c = customerMap.get(cif);
    const ovdDays = Number(row.OVD_DAYS) || 0;
    const revOvdDays = Number(row.REV_OVD_DAYS) || 0;
    const iOvd = Number(row.I_OVD) || 0;
    const pOvd = Number(row.P_OVD) || 0;
    const outstandingLoan = Number(row.OUTSTANDING_LOAN) || 0;
    const minBalNpa = Number(row.MIN_BAL_NPA) || 0;
    const minBalWl = Number(row.MIN_BAL_WL) || 0;
    const opBalance = Number(row.OP_BALANCE) || 0;
    
    c.maxOvdDays = Math.max(c.maxOvdDays, ovdDays);
    c.maxRevOvdDays = Math.max(c.maxRevOvdDays, revOvdDays);
    c.outstandingLoan += outstandingLoan;
    
    // Category calculation: P_OVD inclusion controlled by CATEGORY_CONFIG
    c.netOverdueForCategory += iOvd + (CATEGORY_CONFIG.includePOVDForExpired || !isExpired ? pOvd : 0);
    
    // Display calculation: Exclude P_OVD if EXPIRED
    if (isExpired) {
      c.netOverdueForDisplay += iOvd;
    } else {
      c.netOverdueForDisplay += iOvd + pOvd;
    }
    
    c.minBalNpa = Math.max(c.minBalNpa, minBalNpa);
    c.minBalWl = Math.max(c.minBalWl, minBalWl);
    if (opBalance > 0) c.opBalance = opBalance;
    
    const accNo = row.ACCOUNT_NO;
    if (accNo && !c.accounts.has(accNo)) {
      const totOvdAmt = Number(row.TOT_OVD_AMT) || 0;
      const penal = Number(row.PENAL_INT) || 0;
      
      // For account display: Exclude P_OVD if EXPIRED
      let adjustedTotOvd = totOvdAmt;
      if (isExpired) {
        adjustedTotOvd = totOvdAmt - pOvd;
      }
      
      c.accounts.set(accNo, {
        accountNo: accNo,
        scheme: row.SCHEME || "N/A",
        outstandingLoan: outstandingLoan,
        fullRate: row.FULL_RATE || 0,
        minBalNpa: minBalNpa,
        minBalWl: minBalWl,
        totOvdAmt: adjustedTotOvd,
        penalInt: penal,
        renewal: renewal,
        relationship: row.CUST_RELATIONSHIP || row.RELATIONSHIP || getRowValueInsensitive(row, 'relationship') || getRowValueInsensitive(row, 'custrelationship') || 'Branch'
      });
    }
  });
  Logger.log('RM matches found: ' + rmMatchCount + ' out of ' + customerMap.size + ' unique customers');
  
  const customers = [];
  const allMetrics = [];
  const groupSummaries = new Map();
  
  customerMap.forEach((c, cif) => {
    // Calculate categories using netOverdueForCategory (always includes P_OVD)
    let actualCat = "Pass", probableCat = "Pass";
    if (Math.abs(c.netOverdueForCategory) >= 0.0001) {
      for (let rule of criteria) {
        if (c.maxOvdDays >= rule.min && c.maxOvdDays <= rule.max) actualCat = rule.actual;
        if (c.maxRevOvdDays >= rule.min && c.maxRevOvdDays <= rule.max) probableCat = rule.probable;
      }
    }
    
    const accounts = Array.from(c.accounts.values()).map(a => ({
      ACCOUNT_NO: a.accountNo,
      SCHEME: a.scheme,
      OUTSTANDING_LOAN: a.outstandingLoan,
      FULL_RATE: a.fullRate,
      MIN_BAL_NPA: a.minBalNpa,
      MIN_BAL_WL: a.minBalWl,
      TOT_OVD_AMT: a.totOvdAmt,
      PENAL_INT: a.penalInt,
      NET_OVD: a.totOvdAmt - a.penalInt,
      RELATIONSHIP: a.relationship || c.relationship || 'Branch'
    }));
    
    const totalOvdAmt = accounts.reduce((sum, a) => sum + a.TOT_OVD_AMT, 0);
    
    const metrics = {
      CIF_ID: cif,
      CUSTOMER_NAME: c.name,
      CUSTOMER_GROUP: c.group,
      RENEWAL: c.renewal,
      RELATIONSHIP: c.relationship || 'Branch',
      ASSIGNED_RM: c.assignedRM || 'Unassigned',
      CATEGORY_ACTUAL: actualCat,
      CATEGORY_PROBABLE: probableCat,
      MAX_OVD_DAYS: c.maxOvdDays,
      OUTSTANDING_LOAN: c.outstandingLoan,
      TOTAL_OVERDUE_NET: c.netOverdueForDisplay,  // Display version
      TOT_OVD_AMT: totalOvdAmt,
      MIN_BAL_NPA: c.minBalNpa,
      MIN_BAL_WL: c.minBalWl,
      OP_BALANCE: c.opBalance,
      accounts: accounts
    };
    
    allMetrics.push(metrics);
    customers.push({ id: cif, name: c.name });
    
    if (c.group && c.group.toUpperCase() !== "NO GROUP") {
      if (!groupSummaries.has(c.group)) {
        groupSummaries.set(c.group, {
          groupName: c.group,
          unitCount: 0,
          totalLoan: 0,
          totalOvd: 0,
          totalNetOvd: 0,
          minNpa: 0,
          minWl: 0,
          units: []
        });
      }
      const g = groupSummaries.get(c.group);
      g.unitCount++;
      g.totalLoan += c.outstandingLoan;
      g.totalOvd += totalOvdAmt;
      g.totalNetOvd += c.netOverdueForDisplay;  // Display version
      g.minNpa += c.minBalNpa;
      g.minWl += c.minBalWl;
      g.units.push({
        name: c.name,
        actual: actualCat,
        probable: probableCat,
        totalOvd: totalOvdAmt,
        totalNetOvd: c.netOverdueForDisplay,
        minNpa: c.minBalNpa,
        minWl: c.minBalWl,
        maxOvdDays: c.maxOvdDays
      });
    }
  });
  
  const groups = [];
  groupSummaries.forEach(g => {
    groups.push({
      summary: {
        groupName: g.groupName,
        unitCount: g.unitCount,
        totalLoan: g.totalLoan,
        totalOvd: g.totalOvd,
        totalNetOvd: g.totalNetOvd,
        minNpa: g.minNpa,
        minWl: g.minWl
      },
      units: g.units
    });
  });
  
  const executionTime = new Date().getTime() - startTime;
  Logger.log(`Full data loaded in ${executionTime}ms - ${customers.length} customers, ${groups.length} groups`);
  
  return {
    customers: customers,
    allMetrics: allMetrics,
    groups: groups,
    timestamp: new Date().getTime(),
    executionTime: executionTime,
    totalCustomers: customers.length,
    totalGroups: groups.length
  };
}

// ========== CATEGORY TABLE ENDPOINT ==========
function getCustomersByCategory(category, relationship, sortBy, sortOrder, page, pageSize) {
  category = category || 'ALL';
  relationship = relationship || 'ALL';
  sortBy = sortBy || 'name';
  sortOrder = sortOrder || 'asc';
  page = parseInt(page) || 1;
  pageSize = parseInt(pageSize) || 10;

  const data = getSheetData();
  const criteria = getCategoryCriteria();

  const custMap = new Map();

  data.forEach(row => {
    const cif = row.CIF_ID;
    if (!cif) return;

    if (!custMap.has(cif)) {
      custMap.set(cif, {
        cifId: cif,
        name: row.CUSTOMER_NAME || 'N/A',
        relationship: getRowValueInsensitive(row, 'relationship') || getRowValueInsensitive(row, 'custrelationship') || '',
        maxOvdDays: 0,
        maxRevOvdDays: 0,
        outstandingLoan: 0,
        actualCat: 'Pass',
        probableCat: 'Pass',
        netOverdueForCategory: 0  // Always includes P_OVD
      });
    }

    const cust = custMap.get(cif);
    const ovd = Number(row.OVD_DAYS) || 0;
    const revOvd = Number(row.REV_OVD_DAYS) || 0;
    if (ovd > cust.maxOvdDays) cust.maxOvdDays = ovd;
    if (revOvd > cust.maxRevOvdDays) cust.maxRevOvdDays = revOvd;

    const iOvd = Number(row.I_OVD) || 0;
    const pOvd = Number(row.P_OVD) || 0;
    const renewal = String(row.RENEWAL || '').trim().toUpperCase();
    // Category calculation: P_OVD inclusion controlled by CATEGORY_CONFIG
    cust.netOverdueForCategory += iOvd + (CATEGORY_CONFIG.includePOVDForExpired || renewal !== 'EXPIRED' ? pOvd : 0);

    cust.outstandingLoan += Number(row.OUTSTANDING_LOAN) || 0;
  });

  const customers = [];
  custMap.forEach(cust => {
    // Use netOverdueForCategory for category determination
    if (Math.abs(cust.netOverdueForCategory) < 0.0001) {
      cust.actualCat = 'Pass';
      cust.probableCat = 'Pass';
    } else {
      for (let rule of criteria) {
        if (cust.maxOvdDays >= rule.min && cust.maxOvdDays <= rule.max) {
          cust.actualCat = rule.actual;
          break;
        }
      }
      for (let rule of criteria) {
        if (cust.maxRevOvdDays >= rule.min && cust.maxRevOvdDays <= rule.max) {
          cust.probableCat = rule.probable;
          break;
        }
      }
    }

    if (relationship !== 'ALL') {
      const custRel = (cust.relationship || '').trim().toUpperCase();
      const filterRel = relationship.toUpperCase();
      if (!custRel.includes(filterRel) && !filterRel.includes(custRel)) return;
    }

    let include = false;
    switch (category.toUpperCase()) {
      case 'ALL': include = true; break;
      case 'NPA': include = cust.actualCat.toUpperCase() === 'NPA'; break;
      case 'WL': include = cust.actualCat.toUpperCase() === 'WL'; break;
      case 'PWL': include = cust.probableCat.toUpperCase() === 'PWL'; break;
      case 'PNPA': include = cust.probableCat.toUpperCase() === 'PNPA'; break;
      case 'PASS': include = cust.actualCat.toUpperCase() === 'PASS'; break;
      default: include = true;
    }
    if (!include) return;

    customers.push({
      cifId: cust.cifId,
      name: cust.name,
      limit: cust.outstandingLoan,
      relationship: cust.relationship,
      actualCat: cust.actualCat,
      probableCat: cust.probableCat
    });
  });

  customers.sort((a, b) => {
    let valA, valB;
    if (sortBy === 'name') {
      valA = a.name.toUpperCase();
      valB = b.name.toUpperCase();
    } else {
      valA = a.limit;
      valB = b.limit;
    }
    if (valA < valB) return sortOrder === 'asc' ? -1 : 1;
    if (valA > valB) return sortOrder === 'asc' ? 1 : -1;
    return 0;
  });

  const total = customers.length;
  const start = (page - 1) * pageSize;
  const end = Math.min(start + pageSize, total);
  const paginated = customers.slice(start, end);

  return {
    customers: paginated,
    total: total,
    page: page,
    pageSize: pageSize,
    totalPages: Math.ceil(total / pageSize)
  };
}

function uploadImage(params) {
  if (!params || !params.cifId || !params.imageData) {
    return { success: false, error: 'Missing parameters' };
  }

  const props = PropertiesService.getScriptProperties();
  let pat = props.getProperty('GITHUB_PAT');
  let owner = props.getProperty('GITHUB_OWNER');
  let repo = props.getProperty('GITHUB_REPO');

  if (!pat) return { success: false, error: 'GitHub PAT not configured in script properties' };
  if (!owner) owner = 'user';
  if (!repo) repo = 'bs-main';

  const folder = 'crm-images';
  const ext = 'webp';
  const fileName = params.fileName || `${params.cifId}_${new Date().getTime()}.${ext}`;

  const url = `https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}/contents/${encodeURIComponent(folder)}/${encodeURIComponent(fileName)}`;

  const options = {
    method: 'put',
    headers: {
      'Authorization': 'Bearer ' + pat,
      'Content-Type': 'application/json'
    },
    payload: JSON.stringify({
      message: `Upload CRM image ${fileName}`,
      content: params.imageData,
      branch: 'main'
    }),
    muteHttpExceptions: true
  };

  try {
    const response = UrlFetchApp.fetch(url, options);
    const code = response.getResponseCode();
    const result = JSON.parse(response.getContentText());

    if (code < 200 || code >= 300) {
      return { success: false, error: result.message || 'GitHub API error' };
    }

    const downloadUrl = result.content.download_url;

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName('Images');
    if (!sheet) {
      sheet = ss.insertSheet('Images');
      sheet.getRange(1, 1, 1, 4).setValues([['CIF_ID', 'FileName', 'URL', 'TimeStamp']]);
      sheet.getRange(1, 1, 1, 4).setFontWeight('bold');
    }
    sheet.appendRow([params.cifId, fileName, downloadUrl, new Date()]);

    return { success: true, url: downloadUrl, fileName: fileName };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function deleteImage(fileName) {

  const props = PropertiesService.getScriptProperties();
  let pat = props.getProperty('GITHUB_PAT');
  let owner = props.getProperty('GITHUB_OWNER');

        message: 'Delete CRM image ' + fileName,

        branch: 'main'
      }),
      muteHttpExceptions: true
    };

    UrlFetchApp.fetch(delUrl, delOptions);

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('Images');
    if (sheet) {
      const data = sheet.getDataRange().getValues();

  }

        row.push('');
      }
    });
  }

  const targetCif = String(params.cifId).trim();

  };
  
  if (matchRowIndex === -1) {
    const newRow = new Array(headers.length).fill('');
    newRow[cifIndex] = targetCif;

var FIRESTORE_PROJECT = 'behind-schedule3';

function toFirestoreValue(val) {
  if (val === null || val === undefined) return { nullValue: null };
  if (typeof val === 'string') return { stringValue: val };
  if (typeof val === 'number') {
    if (Number.isInteger(val) && Math.abs(val) < 2e53) return { integerValue: String(val) };
    return { doubleValue: val };
  }
  if (typeof val === 'boolean') return { booleanValue: val };
  if (val instanceof Date) return { timestampValue: val.toISOString() };
  if (Array.isArray(val)) return { arrayValue: { values: val.map(toFirestoreValue)} };
  if (typeof val === 'object') {
    var fields = {};
    Object.keys(val).forEach(function(k) { fields[k] = toFirestoreValue(val[k]); });
    return { mapValue: { fields: fields } };
  }
  return { stringValue: String(val) };
}

function buildFirestoreDocument(obj) {
  var fields = {};
  Object.keys(obj).forEach(function(k) { fields[k] = toFirestoreValue(obj[k]); });
  return { fields: fields };
}

function buildDocPath(collection, docId) {
  return 'projects/' + FIRESTORE_PROJECT + '/databases/(default)/documents/' +
    encodeURIComponent(collection) + '/' + encodeURIComponent(docId);
}

function firestoreBatchCommit(writes) {
  var token = ScriptApp.getOAuthToken();
  var url = 'https://firestore.googleapis.com/v1/projects/' + FIRESTORE_PROJECT +
    '/databases/(default)/documents:commit';
  var options = {
    method: 'POST',
    headers: { 'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json' },
    payload: JSON.stringify({ writes: writes }),
    muteHttpExceptions: true
  };
  var resp = UrlFetchApp.fetch(url, options);
  var code = resp.getResponseCode();
  if (code < 200 || code >= 300) {
    throw new Error('Firestore commit failed: ' + code + ' ' + resp.getContentText());
  }
}

function firestoreDelete(collection, docId) {
  var token = ScriptApp.getOAuthToken();
  var url = 'https://firestore.googleapis.com/v1/projects/' + FIRESTORE_PROJECT +
    '/databases/(default)/documents/' + encodeURIComponent(collection) + '/' + encodeURIComponent(docId);
  var options = {
    method: 'DELETE',
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  };
  UrlFetchApp.fetch(url, options);
}

function syncAll() {
  var start = new Date().getTime();
  var syncToken = ScriptApp.getOAuthToken();
  var baseUrl = 'https://firestore.googleapis.com/v1/projects/' + FIRESTORE_PROJECT + '/databases/(default)/documents/';

  var wipeCollections = ['customers', 'metrics', 'groups', 'insights', 'customerInfo'];

  var deletedCount = 0;
  wipeCollections.forEach(function(coll) {
    var pageToken = '';
    do {
      var url = baseUrl + encodeURIComponent(coll) + '?pageSize=300' + (pageToken ? '&pageToken=' + encodeURIComponent(pageToken) : '');
      var resp = UrlFetchApp.fetch(url, {
        method: 'GET',
        headers: { 'Authorization': 'Bearer ' + syncToken },
        muteHttpExceptions: true
      });
      var data = JSON.parse(resp.getContentText());
      if (data.documents) {
        data.documents.forEach(function(doc) {
          UrlFetchApp.fetch(doc.name + '?allowMissing=true', {
            method: 'DELETE',
            headers: { 'Authorization': 'Bearer ' + syncToken },
            muteHttpExceptions: true
          });
          deletedCount++;
        });
      }
      pageToken = data.nextPageToken || '';
    } while (pageToken);
  });

  var syncResult = syncToFirestore();

  var elapsed = new Date().getTime() - start;
  Logger.log('syncAll complete: deleted ' + deletedCount + ' docs, synced ' + syncResult.customers + ' customers in ' + elapsed + 'ms');
  return {
    success: true,
    deleted: deletedCount,
    synced: syncResult,
    elapsed: elapsed
  };
}
function syncToFirestore() {
  Logger.log('Starting full Firestore sync...');
  var start = new Date().getTime();

  var full = loadFullData();
  if (!full || !full.allMetrics) throw new Error('loadFullData returned no data');

  var writes = [];
  function qw(collection, docId, data) {
    writes.push({ update: { name: buildDocPath(collection, docId), fields: buildFirestoreDocument(data).fields } });
    if (writes.length >= 500) { firestoreBatchCommit(writes); writes = []; }
  }
  function flush() { if (writes.length > 0) { firestoreBatchCommit(writes); writes = []; } }

  var customerCount = 0;
  full.allMetrics.forEach(function(m) {
    var cif = String(m.CIF_ID).trim().replace(/^0+/, '');
    if (!cif) return;

    qw('customers', cif, {
      name: m.CUSTOMER_NAME || 'N/A',
      cifId: cif,
      categoryActual: m.CATEGORY_ACTUAL || 'Pass',
      categoryProbable: m.CATEGORY_PROBABLE || 'Pass',
      renewal: m.RENEWAL || 'N/A',
      outstandingLoan: m.OUTSTANDING_LOAN || 0,
      totOvdAmt: m.TOT_OVD_AMT || 0,
      totalOverdueNet: m.TOTAL_OVERDUE_NET || 0,
      minBalNpa: m.MIN_BAL_NPA || 0,
      minBalWl: m.MIN_BAL_WL || 0,
      maxOvdDays: m.MAX_OVD_DAYS || 0,
      opBalance: m.OP_BALANCE || 0,
      customerGroup: m.CUSTOMER_GROUP || 'NO GROUP',
      relationship: m.RELATIONSHIP || 'Branch',
      assignedRm: m.ASSIGNED_RM || 'Unassigned'
    });

    qw('metrics', cif, m);
    customerCount++;
  });

  var groupCount = 0;
  if (full.groups) {
    full.groups.forEach(function(g) {
      var groupKey = g.summary.groupName.replace(/[^a-zA-Z0-9_-]/g, '_');
      qw('groups', groupKey, {
        groupName: g.summary.groupName,
        unitCount: g.summary.unitCount,
        totalLoan: g.summary.totalLoan,
        totalOvd: g.summary.totalOvd,
        totalNetOvd: g.summary.totalNetOvd,
        minNpa: g.summary.minNpa,
        minWl: g.summary.minWl,
        units: g.units
      });
      groupCount++;
    });
  }

  var contactCount = 0;
  var ciSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('CustomerInfo');
  if (ciSheet) {
    var ciData = ciSheet.getDataRange().getDisplayValues();
    if (ciData.length > 1) {
      var ciHeaders = ciData[0].map(function(h) { return String(h).trim(); });
      var cifIdx = ciHeaders.findIndex(function(h) {
        var norm = h.toLowerCase().replace(/[\s_-]/g, '');
        return norm === 'cifid' || norm === 'cif';
      });
      if (cifIdx !== -1) {
        for (var i = 1; i < ciData.length; i++) {
          var ciRow = ciData[i];
          var ciCif = String(ciRow[cifIdx]).trim().replace(/^0+/, '');
          if (!ciCif) continue;
          var ciObj = {};
          ciHeaders.forEach(function(h, idx) { ciObj[h] = ciRow[idx]; });
          qw('customerInfo', ciCif, ciObj);
          contactCount++;
        }
      }
    }
  }

  flush();

  var elapsed = new Date().getTime() - start;
  Logger.log('Firestore sync complete: ' + customerCount + ' customers, ' + groupCount + ' groups, ' + contactCount + ' contacts in ' + elapsed + 'ms');
  return { success: true, customers: customerCount, groups: groupCount, contacts: contactCount, elapsed: elapsed };
}

function onEdit(e) {
  if (!e || !e.range) return;
  var sheet = e.range.getSheet();
  var sheetName = sheet ? sheet.getName() : '';

  // Only sync sheets that affect Firestore data
  var syncSheets = ['Data', 'CustomerInfo'];
  if (syncSheets.indexOf(sheetName) === -1) return;

  // Debounce: use LockService to avoid concurrent runs
  var lock = LockService.getDocumentLock();
  if (!lock.tryLock(5000)) return;

  try {
    // For small edits, just trigger a full re-sync
    // (Incremental sync would require tracking which CIFs changed)
    syncToFirestore();
  } catch(e) {
    Logger.error('onEdit sync failed: ' + e);
  } finally {
    lock.releaseLock();
  }
}
